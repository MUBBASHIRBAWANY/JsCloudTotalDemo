export const createVoucher = async (req, res) => {


    try {
        const token = req.headers.authorization?.split(" ")[1];
        const decoded = jwt.verify(token, "peral-Dynamic");
        const createdByUser = decoded?.val?.firstname || "unknown";
        const userId = req.user?._id || req.user?.id; // Adjust based on your auth setup


        const {
            VoucherNumber,
            ChequeNumber,
            VoucherType,
            Type,
            VoucharData
        } = req.body;

        // 🔒 Voucher uniqueness
        const existingVoucher = await VoucherModal.findOne({ VoucherNumber });
        if (existingVoucher) {

            return res.status(400).json({ message: "Voucher number already exists" });
        }

        // 🔒 Cheque uniqueness
        if (ChequeNumber) {

            if (ChequeNumber !== "Online" && VoucherType == "BP") {
                // console.log(ChequeNumber)
                const existingChq = await VoucherModal.findOne({ ChequeNumber });
                if (existingChq) {

                    return res
                        .status(400)
                        .json({ message: `Cheque already used in ${existingChq.VoucherNumber}` });
                }
                else {
                    res.status(201).json({
                        message: "Voucher created successfully",

                    });
                }
            }

        }

        // 🟡 STEP 1: VALIDATE invoices ONLY
        if (VoucherType === "CR") {

            for (const item of VoucharData.filter(v => v.Client)) {

                const invoiceRef = item.ClientRef2 || "";
                const amount = Number(item.Credit);
                if (invoiceRef !== "") {
                    const invoice = await SalesInvoiceModal
                        .findOne({ SalesInvoice: invoiceRef })
                        ;

                    if (invoice) {
                        if (Number(invoice.RemainingAmount) < amount) {
                            throw new Error(
                                `Insufficient balance in invoice ${invoiceRef}`
                            );
                        }
                        else {
                            await SalesInvoiceModal.findByIdAndUpdate(invoice._id, {
                                RemainingAmount: Number(invoice.RemainingAmount) - amount,
                                RecivedAmount: Number(invoice.RecivedAmount) + amount,
                            })
                        }

                    } else {
                        const opening = await InvoiceOpeningModal.findOne();
                        const inv = opening?.InvoiceData.find(i => i.Invoice === invoiceRef);

                        if (!inv) {
                            throw new Error(`Invoice not found: ${invoiceRef}`);
                        }

                        if (Number(inv.RemaingAmount) < amount) {
                            throw new Error(
                                `Insufficient opening balance in invoice ${invoiceRef}`
                            );
                        }


                    }
                }

            }
            const newVoucher = await VoucherModal.create(req.body);
            await logModal.create({
                userId: userId,
                action: "create",
                Page : "Voucher",
                payload: req.body,
                responseStatus: 200,
                RefNumber: VoucherNumber,
            });
            res.status(201).json({
                message: "Voucher created successfully",
                voucher: newVoucher
            });
        }


        // 🟢 STEP 2: CREATE voucher FIRST
        else if ((Type !== "Online" || Type !== "Cash")) {
            const newVoucher = await VoucherModal.create(req.body);
            await logModal.create({
                userId: userId,
                action: "create",
                Page : "Voucher",
                payload: req.body,
                responseStatus: 200,
                RefNumber: VoucherNumber,
            });
            res.status(201).json({
                message: "Voucher created successfully",
                voucher: newVoucher
            });
        }
        else {
            for (const item of VoucharData.filter(v => v.Client)) {

                const invoiceRef = item.ClientRef2;
                const amount = Number(item.Credit);

                const invoice = await SalesInvoiceModal
                    .findOne({ SalesInvoice: invoiceRef })


                if (invoice) {
                    invoice.RecivedAmount += amount;
                    invoice.RemainingAmount -= amount;
                    await invoice.save();
                } else {
                    const opening = await InvoiceOpeningModal.findOne();
                    const inv = opening.InvoiceData.find(i => i.Invoice === invoiceRef);

                    inv.RecivedAmount = (Number(inv.RecivedAmount) + amount).toString();
                    inv.RemaingAmount = (Number(inv.RemaingAmount) - amount).toString();

                    await InvoiceOpeningModal.findByIdAndUpdate(opening._id, opening)
                }
            }

            // ✅ COMMIT

            const newVoucher = await VoucherModal.create(req.body);
            await logModal.create({
                userId: userId,
                action: "create",
                Page : "Voucher",
                payload: req.body,
                responseStatus: 200,
                RefNumber: VoucherNumber,
            });

            res.status(201).json({
                message: "Voucher created successfully",
                voucher: newVoucher
            });
        }
        // 🟢 STEP 3: UPDATE invoices AFTER voucher success


    } catch (error) {

        console.log(error)
        res.status(400).json({
            message: error.message
        });
    }
};

export const PushVoucherBulkData = async (req, res) => {
    try {
        const insertedIds = [];
        for (const item of req.body) {

            const voucher = await VoucherModal.find({ VoucherType: item.VoucherType }).sort({ _id: -1 }).limit(1)
            // console.log(voucher);
            if (voucher.length > 0) {
                let nextVoucherNumber = (parseInt(voucher[0].VoucherNumber.slice('2', "9"))) + 1;
                item.VoucherNumber = item.VoucherType + nextVoucherNumber.toString().padStart(7, '0');
            }
            else {
                item.VoucherNumber = `${item.VoucherType}0000001`; // Start with 1 if no vouchers exist
            }
            const newVoucher = new VoucherModal({
                VoucherType: item.VoucherType,
                VoucherNumber: item.VoucherNumber,
                VoucherDate: item.VoucherDate,
                VoucherMainAccount: item.VoucherMainAccount,
                VoucharData: item.VoucharData,
            });
            const savedItem = await newVoucher.save();
            insertedIds.push(savedItem._id);
            console.log(`Inserted: ${item.VoucherType}`);
        }
        res.status(200).send({
            status: true,
            data: "Data Added"
        });
        console.log('All vouchers inserted successfully!');
        return insertedIds;
    } catch (error) {
        console.error('Error inserting vouchers:', error.message);
        throw error;
    }
}

export const createRealizeVoucher = async (req, res) => {
    try {
        const token = req.headers.authorization?.split(" ")[1];
        const decoded = jwt.verify(token, "peral-Dynamic");
        const createdByUser = decoded?.val?.firstname || "unknown";
        const { VoucharData } = req.body;

        // Step 1: Extract client items
        const OnlyClient = VoucharData.filter((item) => item.hasOwnProperty("Client"));
        const onlyCHq = await Promise.all(
            OnlyClient.map(async (item) => {
                const invoice = await SalesInvoiceModal.findOne({ SalesFlowRef: item.ClientRef2 })
                const ref =
                    item.Ref ||
                    invoice?.SalesInvoice;
                const client = await CutomerModal.findOne({ SalesFlowRef: item.Client })
                return {
                    rowid: item.id,
                    ClientRef2: invoice?.SalesInvoice || item.ClientRef2,
                    Store: invoice?.Store,
                    ref: ref,
                    Client: client?._id.toString(),
                    Credit: Number(item.Credit) || 0,
                    Type: item.Type, // "Current" or "Opening",
                    Narration: invoice?.SalesInvoice
                        ? `Cash received against invoice ${invoice?.SalesInvoice} from ${(client?.CutomerName || "Unknown Client")}`
                        : item.Narration || ""
                };
            })
        );

        const errors = [];

        // Step 2: Validate before save
        for (const item of onlyCHq) {
            let customerInvoice;
            let openingDoc;

            if (item.Type === "Current") {
                customerInvoice = await SalesInvoiceModal.findOne({ SalesInvoice: item.ref });
            } else if (item.Type === "Opening") {
                openingDoc = await InvoiceOpeningModal.findOne({
                    "InvoiceData.Invoice": item.ref,
                });
                customerInvoice = openingDoc?.InvoiceData?.find(
                    (inv) => inv.Invoice === item.ref
                );
            }

            if (!customerInvoice) {
                errors.push(`Invoice not found for ref: ${item.ref} (${item.Type})`);
                continue;
            }

            const remaining = Number(customerInvoice.RemaingAmount || customerInvoice.RemainingAmount) || 0;
            const credit = Number(item.Credit) || 0;

            if (credit > remaining) {
                errors.push(
                    `Credit amount (${credit}) exceeds remaining balance (${remaining}) for invoice ${item.ref} (${item.Type})`
                );
            }

            if (remaining <= 0) {
                errors.push(`Invoice ${item.ref} (${item.Type}) is already fully paid.`);
            }
        }

        if (errors.length > 0) {
            return res.status(400).json({
                success: false,
                message: "Validation failed before posting voucher.",
                errors,
            });
        }

        // Step 3: Generate new voucher number
        const lastVoucher = await VoucherModal.find({ VoucherType: "CR" })
            .sort({ _id: -1 })
            .limit(1);

        let VoucherNumber;
        if (lastVoucher.length === 0) {
            VoucherNumber = `CR0000001`;
        } else {
            const nextVoucherNumber =
                parseInt(lastVoucher[0].VoucherNumber.slice(2)) + 1;
            VoucherNumber = `CR${nextVoucherNumber.toString().padStart(7, "0")}`;
        }
        onlyCHq.push({
            Debit: onlyCHq.reduce((a, b) => a + (b.Credit || 0), 0),
            Account: req.body.VoucherMainAccount,
            store: onlyCHq[0].Store,
            show: true,
            id: Date.now(),
            Narration: "Petty Cash Account",
        });
        // Step 4: Save voucher
        const Voucher = {
            ...req.body,
            VoucharData: onlyCHq,
            VoucherNumber,

        };

        const newVoucher = new VoucherModal(Voucher);

        await VoucherLogModel.create({
            VoucherNumber: newVoucher.VoucherNumber,
            createdBy: createdByUser,
            createdAt: new Date().toLocaleString("en-US", {
                hour: "2-digit",
                minute: "2-digit",
                hour12: true,
                day: "2-digit",
                month: "short",
                year: "numeric",
            }),
        });
        await newVoucher.save();

        // Step 5: Update related invoices
        for (const item of onlyCHq) {
            if (item.Type === "Current") {
                const customerInvoice = await SalesInvoiceModal.findOne({ SalesInvoice: item.ref });

                const remaining = Number(customerInvoice.RemainingAmount) || 0;
                const received = Number(customerInvoice.RecivedAmount) || 0;
                const credit = Number(item.Credit) || 0;

                const newRemaining = remaining - credit;
                const newReceived = received + credit;

                if (!Array.isArray(customerInvoice.PaymentDetail)) {
                    customerInvoice.PaymentDetail = [];
                }

                customerInvoice.PaymentDetail.push({
                    type: "Cash",
                    Amount: credit,
                });

                await SalesInvoiceModal.findByIdAndUpdate(
                    customerInvoice._id,
                    {
                        RemainingAmount: newRemaining,
                        RecivedAmount: newReceived,
                        PaymentDetail: customerInvoice.PaymentDetail,
                    },
                    { new: true }
                );
            }

            // 🔹 Handle Opening type
            if (item.Type === "Opening") {
                const openingDoc = await InvoiceOpeningModal.findOne({
                    "InvoiceData.Invoice": item.ref,
                });

                if (!openingDoc) continue;

                const invoiceIndex = openingDoc.InvoiceData.findIndex(
                    (inv) => inv.Invoice === item.ref
                );

                if (invoiceIndex === -1) continue;

                const current = openingDoc.InvoiceData[invoiceIndex];
                const remaining = Number(current.RemaingAmount) || 0;
                const received = Number(current.RecivedAmount) || 0;
                const credit = Number(item.Credit) || 0;

                const newRemaining = remaining - credit;
                const newReceived = received + credit;

                openingDoc.InvoiceData[invoiceIndex].RemaingAmount = newRemaining;
                openingDoc.InvoiceData[invoiceIndex].RecivedAmount = newReceived;

                if (!Array.isArray(openingDoc.InvoiceData[invoiceIndex].PaymentDetail)) {
                    openingDoc.InvoiceData[invoiceIndex].PaymentDetail = [];
                }

                openingDoc.InvoiceData[invoiceIndex].PaymentDetail.push({
                    type: "Cash",
                    Amount: credit,
                });

                await openingDoc.save();
            }
        }

        // Step 6: Mark voucher as posted
        const updatedVoucher = await VoucherModal.findByIdAndUpdate(
            newVoucher._id,
            { status: "Post" },
            { new: true }
        );

        res.status(200).json({
            success: true,
            message: "Voucher successfully created and posted.",
            newVoucher,
            updatedVoucher,
        });
    } catch (err) {
        console.error(err);
        res.status(400).json({
            success: false,
            error: err.message,
        });
    }
};

        const { id } = req.params;

        // Validate ID
        if (!id || id === 'undefined' || id === 'null') {
            return res.status(400).json({
                success: false,
                message: "Invalid voucher ID provided"
            });
        }

        // 1. Fetch voucher with proper error handling
        const data = await VoucherModal.findById(id);

        if (!data) {
            return res.status(404).json({
                success: false,
                message: "Voucher not found"
            });
        }

        // 2. Validate VoucharData exists
        if (!data.VoucharData || !Array.isArray(data.VoucharData)) {
            return res.status(200).json({
                success: true,
                data: []
            });
        }

        const VoucharData = data.VoucharData;

        // 3. Extract IDs with better filtering (handle null, undefined, empty strings)
        const AccountID = VoucharData
            .map(item => item?.Account)
            .filter(val => val && val !== 'undefined' && val !== 'null');

        const transporterID = VoucharData
            .map(item => item?.Transporter)
            .filter(val => val && val !== 'undefined' && val !== 'null');

        const VendorID = VoucharData
            .map(item => item?.Vendor)
            .filter(val => val && val !== 'undefined' && val !== 'null');

        const ClientID = VoucharData
            .map(item => item?.Client)
            .filter(val => val && val !== 'undefined' && val !== 'null');

        const isValidObjectId = (id) => {
            return typeof id === "string" && /^[a-fA-F0-9]{24}$/.test(id);
        };
        const StoreID = VoucharData
            .map(item => item?.store ? item.store.toString() : null)
            .filter(id => isValidObjectId(id)); // ✅ only valid IDs
        console.log(StoreID);

        // 4. Optimize: Only query if IDs exist
        const promises = [];

        if (AccountID.length > 0) {
            promises.push(
                ChartofAccountsModel.find({ _id: { $in: AccountID } })
                    .select("AccountName")
                    .lean() // Use lean() for better performance
                    .then(accounts => accounts.map(acc => ({ ...acc, type: 'Account' })))
            );
        } else {
            promises.push(Promise.resolve([]));
        }

        if (transporterID.length > 0) {
            promises.push(
                TransporterModal.find({ _id: { $in: transporterID } })
                    .select("TransporterName")
                    .lean()
                    .then(transporters => transporters.map(t => ({ ...t, type: 'Transporter' })))
            );
        } else {
            promises.push(Promise.resolve([]));
        }

        if (VendorID.length > 0) {
            promises.push(
                VendorModel.find({ _id: { $in: VendorID } })
                    .select("VendorName")
                    .lean()
                    .then(vendors => vendors.map(v => ({ ...v, type: 'Vendor' })))
            );
        } else {
            promises.push(Promise.resolve([]));
        }

        if (ClientID.length > 0) {
            promises.push(
                CutomerModal.find({ _id: { $in: ClientID } })
                    .select("CutomerName")
                    .lean()
                    .then(clients => clients.map(c => ({ ...c, type: 'Client' })))
            );
        } else {
            promises.push(Promise.resolve([]));
        }
        if (StoreID.length > 0) {
            promises.push(
                StoreModal.find({ _id: { $in: StoreID } })
                    .select("StoreName")
                    .lean()
                    .then(stores => stores.map(s => ({ ...s, type: 'Store' })))
            );
        } else {
            promises.push(Promise.resolve([]));
        }


        // 5. Execute all queries in parallel
        const [accounts, transporters, vendors, clients, stores] = await Promise.all(promises);

        // 6. Merge with type information
        const mergeData = [...accounts, ...transporters, ...vendors, ...clients, ...stores];

        // 7. Return structured response
        res.status(200).json({
            success: true,
            data: mergeData,
            voucher: data,
            metadata: {
                voucherId: id,
                totalReferences: mergeData.length,

                breakdown: {
                    accounts: accounts.length,
                    transporters: transporters.length,
                    vendors: vendors.length,
                    clients: clients.length
                }
            }
        });

    } catch (err) {
        console.error("Error in VoucherByID:", {
            message: err.message,
            stack: err.stack,
            params: req.params
        });

        // Send proper error response based on error type
        if (err.name === 'CastError') {
            return res.status(400).json({
                success: false,
                message: "Invalid ID format"
            });
        }

        res.status(500).json({
            success: false,
            message: "Internal server error. Please try again later."
        });
    }
}
